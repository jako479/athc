**Goal:** A read-only library that returns a `.lge` league's conference names, division names and team names (team name, nickname, official name).

**Architecture:** Same shape as `fbpro98_lg2`: `schema.py` constants, frozen slotted dataclasses in `model.py`, `reader.py` with `read_lge(league, league_dir)` / `parse_lge(buffer, path)` and `InvalidLgeError(AthcError)`. The reader walks the chunk tree in file order, checks each chunk's tag and size, and unmasks team chunks.

**Tech Stack:** Python 3.13 stdlib (`struct`, `dataclasses`, `pathlib`), pytest.

**Spec:** `docs/superpowers/specs/2026-10-09-fbpro98-lge-design.md`; format `docs/fbpro98_lge/specs/lge.md`.

- Libraries never print, never log; failures raise an `AthcError` subclass with the reason and the path.

## Review Focus

- A file that is not a `.lge` (another league file, random bytes): rejected naming the first chunk, never a crash. Test: unprintable tag shown as hex.
- A chunk size field far past the file end (u32 max): rejected, no huge allocation. Test: `test_huge_chunk_size_rejected`.

```python
"""Binary layout constants for the FbPro98 .lge league file format.

Shared by the reader and its tests. See specs/lge.md for the full format.
"""

from struct import Struct

CHUNK_HEADER = Struct("<4sI")  # tag, data size

ID_L03 = b"L03:"  # league
ID_C03 = b"C03:"  # conference
ID_D03 = b"D03:"  # division
ID_T03 = b"T03:"  # team, masked
ID_R03 = b"R03:"  # roster, masked
ID_S03 = b"S03:"  # schedule

# L03, C03 and D03 are a fixed part plus one id byte per conference, division
# or team; the count sits in the fixed part.
LEAGUE_BASE_SIZE = 89
LEAGUE_CONFERENCES_OFFSET = 0x01
CONFERENCE_BASE_SIZE = 29
CONFERENCE_DIVISIONS_OFFSET = 0x02
CONFERENCE_NAME_OFFSET = 0x04
DIVISION_BASE_SIZE = 29
DIVISION_TEAMS_OFFSET = 0x03
DIVISION_NAME_OFFSET = 0x04
NAME_SIZE = 25  # conference and division names

TEAM_CHUNK_SIZE = 171
ROSTER_CHUNK_SIZE = 325

# Offsets into the unmasked team bytes, after the clear team number.
TEAM_NAME_OFFSET = 0x36
NICKNAME_OFFSET = 0x47
TEAM_TEXT_SIZE = 17

# Team byte i is XORed with (MASK_FACTOR * team number + i) & 0xFF.
MASK_FACTOR = 0x69
```

```python
"""In-memory data model for FbPro98 .lge league files.

The league's conferences, their divisions and the divisions' teams, nested the
way the file's chunk tree is, in file order.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class Team:
    """One team's names, as on the game's Team Settings screen."""

    name: str
    """The "Team Name", e.g. `Denver`."""

    nickname: str
    """The "Nickname", e.g. `Broncos`."""

    @property
    def official_name(self) -> str:
        """Team name and nickname, e.g. `Denver Broncos`; just the one present
        when the other is empty."""
        return " ".join(part for part in (self.name, self.nickname) if part)


@dataclass(frozen=True, slots=True)
class Division:
    """One division and its teams, in file order."""

    name: str
    teams: tuple[Team, ...]


@dataclass(frozen=True, slots=True)
class Conference:
    """One conference and its divisions, in file order."""

    name: str
    divisions: tuple[Division, ...]


@dataclass(frozen=True, slots=True)
class LgeFile:
    """The parts of a league's `.lge` file this library reads."""

    conferences: tuple[Conference, ...]
    """In file order. Walking conferences, divisions and teams in order gives
    team numbers 1..N, which key the other league files (see specs/lge.md)."""
```

"""Shared fixtures for fbpro98_lge tests.

`build_chunks` builds a valid .lge as a list of chunks; replace, drop or add
one to get the malformed and edge buffers no real file holds, then join them
with `build_lge`. `mask` is the tests' own copy of the game's mask.
"""

from __future__ import annotations

from collections.abc import Sequence

import pytest

from athc.fbpro98_lge.schema import (
    CHUNK_HEADER,
    CONFERENCE_BASE_SIZE,
    CONFERENCE_DIVISIONS_OFFSET,
    CONFERENCE_NAME_OFFSET,
    DIVISION_BASE_SIZE,
    DIVISION_NAME_OFFSET,
    DIVISION_TEAMS_OFFSET,
    ID_C03,
    ID_D03,
    ID_L03,
    ID_R03,
    ID_S03,
    ID_T03,
    LEAGUE_BASE_SIZE,
    LEAGUE_CONFERENCES_OFFSET,
    MASK_FACTOR,
    NICKNAME_OFFSET,
    ROSTER_CHUNK_SIZE,
    TEAM_CHUNK_SIZE,
    TEAM_NAME_OFFSET,
)


def build_chunk(tag: bytes, data: bytes) -> bytes:
    """One chunk: tag, data size, data."""
    return CHUNK_HEADER.pack(tag, len(data)) + data


def build_league_data(conferences: int) -> bytes:
    """L03 data: the conference count; everything else zero."""
    data = bytearray(LEAGUE_BASE_SIZE + conferences)
    data[LEAGUE_CONFERENCES_OFFSET] = conferences
    return bytes(data)


def _counted(base_size: int, count_offset: int, name_offset: int, count: int, name: bytes) -> bytes:
    # Ids 1..count fill the tail, as in real files, so a name that fills its
    # field has text right after it.
    data = bytearray(base_size) + bytes(range(1, count + 1))
    data[count_offset] = count
    data[name_offset : name_offset + len(name)] = name
    return bytes(data)


def build_conference_data(divisions: int, name: bytes = b"") -> bytes:
    """C03 data: division count and name."""
    return _counted(CONFERENCE_BASE_SIZE, CONFERENCE_DIVISIONS_OFFSET, CONFERENCE_NAME_OFFSET, divisions, name)


def build_division_data(teams: int, name: bytes = b"") -> bytes:
    """D03 data: team count and name."""
    return _counted(DIVISION_BASE_SIZE, DIVISION_TEAMS_OFFSET, DIVISION_NAME_OFFSET, teams, name)


def mask(team: int, data: bytes) -> bytes:
    """The game's XOR mask; applying it twice gives the data back."""
    return bytes(byte ^ ((MASK_FACTOR * team + index) & 0xFF) for index, byte in enumerate(data))


def build_team_data(number: int, name: bytes = b"Denver", nickname: bytes = b"Broncos") -> bytes:
    """T03 data: the clear team number, then the masked fields."""
    fields = bytearray(TEAM_CHUNK_SIZE - 1)
    fields[TEAM_NAME_OFFSET : TEAM_NAME_OFFSET + len(name)] = name
    fields[NICKNAME_OFFSET : NICKNAME_OFFSET + len(nickname)] = nickname
    return bytes([number]) + mask(number, bytes(fields))


def build_roster_data(number: int) -> bytes:
    """R03 data: the clear team number (u16), then zeros; not read."""
    return number.to_bytes(2, "little") + bytes(ROSTER_CHUNK_SIZE - 2)


def build_chunks(shape: Sequence[Sequence[int]] = ((1,),)) -> list[bytes]:
    """A valid file's chunks. `shape` lists, per conference, each division's
    team count; teams are numbered 1..N in file order."""
    chunks = [build_chunk(ID_L03, build_league_data(len(shape)))]
    number = 0
    for divisions in shape:
        chunks.append(build_chunk(ID_C03, build_conference_data(len(divisions))))
        for teams in divisions:
            chunks.append(build_chunk(ID_D03, build_division_data(teams)))
            for _ in range(teams):
                number += 1
                chunks.append(build_chunk(ID_T03, build_team_data(number)))
                chunks.append(build_chunk(ID_R03, build_roster_data(number)))
    chunks.append(build_chunk(ID_S03, bytes(3)))
    return chunks


def build_lge(chunks: Sequence[bytes] | None = None) -> bytes:
    """A whole file from its chunks (default: one team)."""
    return b"".join(build_chunks() if chunks is None else chunks)

def test_names_read_up_to_nul(
    make_chunks,
    make_chunk,
    make_conference_data,
    make_division_data,
    make_team_data,
    make_lge,

    chunks = make_chunks()

def test_names_filling_their_field_read_whole(
    make_chunks,
    make_chunk,
    make_conference_data,
    make_division_data,
    make_team_data,
    make_lge,
):
    # No NUL: the name is the whole field and stops there, though the next
    # field's bytes follow at once.
    chunks = make_chunks()
    chunks[C03] = make_chunk(ID_C03, make_conference_data(1, b"C" * NAME_SIZE))
    chunks[D03] = make_chunk(ID_D03, make_division_data(1, b"D" * NAME_SIZE))
    team = make_team_data(1, b"N" * TEAM_TEXT_SIZE, b"K" * TEAM_TEXT_SIZE)
    chunks[T03] = make_chunk(ID_T03, team)
    lge = parse_lge(make_lge(chunks))
    assert lge.conferences[0].name == "C" * 25
    assert lge.conferences[0].divisions[0].name == "D" * 25
    assert _only_team(lge) == Team("N" * 17, "K" * 17)

"""Parse FbPro98 .lge league files into LgeFile objects. See specs/lge.md."""

from __future__ import annotations

from os import PathLike
from pathlib import Path

from athc.errors import AthcError
from athc.fbpro98_lge.model import Conference, Division, LgeFile, Team
from athc.fbpro98_lge.schema import (
    CHUNK_HEADER,
    CONFERENCE_BASE_SIZE,
    CONFERENCE_DIVISIONS_OFFSET,
    CONFERENCE_NAME_OFFSET,
    DIVISION_BASE_SIZE,
    DIVISION_NAME_OFFSET,
    DIVISION_TEAMS_OFFSET,
    ID_C03,
    ID_D03,
    ID_L03,
    ID_R03,
    ID_S03,
    ID_T03,
    LEAGUE_BASE_SIZE,
    LEAGUE_CONFERENCES_OFFSET,
    MASK_FACTOR,
    NAME_SIZE,
    NICKNAME_OFFSET,
    ROSTER_CHUNK_SIZE,
    TEAM_CHUNK_SIZE,
    TEAM_NAME_OFFSET,
    TEAM_TEXT_SIZE,
)

StrPath = str | PathLike[str]


class InvalidLgeError(AthcError):
    """Raised when a `.lge` file is structurally invalid."""


def read_lge(league: str, league_dir: StrPath) -> LgeFile:
    """Read and parse a league's .lge file from disk.

    Args:
        league: League name; the file read is `<league>.lge`.
        league_dir: Folder holding the league's files.

    Returns:
        Parsed LgeFile.

    Raises:
        InvalidLgeError: If the file is empty, a chunk is cut short, missing,
            out of order or the wrong size, or data follows the schedule.
        OSError: If the file cannot be opened or read.
    """
    file_path = Path(league_dir) / f"{league}.lge"
    return parse_lge(file_path.read_bytes(), file_path)


def parse_lge(buffer: bytes, path: StrPath = "<buffer>") -> LgeFile:
    """Parse a .lge league file from raw bytes.

    Args:
        buffer: Full contents of a .lge file.
        path: Path used only in error messages. Defaults to "<buffer>" when
            parsing data that did not come from disk.

    Returns:
        Parsed LgeFile.

    Raises:
        InvalidLgeError: If the buffer is empty, a chunk is cut short, missing,
            out of order or the wrong size, or data follows the schedule.
    """
    file_path = Path(path)
    if not buffer:
        raise InvalidLgeError("Empty file", file_path)
    chunks = _Chunks(buffer, file_path)
    _, conference_count = chunks.take_counted(
        ID_L03, LEAGUE_BASE_SIZE, LEAGUE_CONFERENCES_OFFSET
    )
    conferences = tuple(_parse_conference(chunks) for _ in range(conference_count))
    # The schedule is not read, only required, so a tree with more conferences,
    # divisions or teams than its counts say fails here.
    chunks.take(ID_S03)
    chunks.finish()
    return LgeFile(conferences=conferences)


def _parse_conference(chunks: _Chunks) -> Conference:
    data, division_count = chunks.take_counted(
        ID_C03, CONFERENCE_BASE_SIZE, CONFERENCE_DIVISIONS_OFFSET
    )
    name = _read_text(data, CONFERENCE_NAME_OFFSET, NAME_SIZE)
    divisions = tuple(_parse_division(chunks) for _ in range(division_count))
    return Conference(name=name, divisions=divisions)


def _parse_division(chunks: _Chunks) -> Division:
    data, team_count = chunks.take_counted(
        ID_D03, DIVISION_BASE_SIZE, DIVISION_TEAMS_OFFSET
    )
    name = _read_text(data, DIVISION_NAME_OFFSET, NAME_SIZE)
    teams = tuple(_parse_team(chunks) for _ in range(team_count))
    return Division(name=name, teams=teams)


def _parse_team(chunks: _Chunks) -> Team:
    data = chunks.take(ID_T03, TEAM_CHUNK_SIZE)
    # Byte 0 is the team number, in clear; it keys the mask on the rest.
    fields = _unmask(data[0], data[1:])
    # Every team is followed by its roster, which is not read.
    chunks.take(ID_R03, ROSTER_CHUNK_SIZE)
    return Team(
        name=_read_text(fields, TEAM_NAME_OFFSET, TEAM_TEXT_SIZE),
        nickname=_read_text(fields, NICKNAME_OFFSET, TEAM_TEXT_SIZE),
    )


def _unmask(team: int, data: bytes) -> bytes:
    return bytes(
        byte ^ ((MASK_FACTOR * team + index) & 0xFF) for index, byte in enumerate(data)
    )


def _read_text(data: bytes, offset: int, size: int) -> str:
    # Text after the NUL is left over from an earlier, longer value; a name
    # that fills its field has no NUL.
    field = data[offset : offset + size]
    end = field.find(b"\x00")
    return field[: size if end == -1 else end].decode("ASCII", errors="replace")


def _tag_name(tag: bytes) -> str:
    # A tag is four printable ASCII characters ending in a colon; anything
    # else, from a file that is not a .lge, shows as hex.
    text = tag.decode("latin-1")
    if tag.isascii() and text.isprintable():
        return text.removesuffix(":")
    return tag.hex(" ")


class _Chunks:
    """Hands out the file's chunks in order, checking each tag and size."""

    def __init__(self, buffer: bytes, path: Path) -> None:
        self._buffer = buffer
        self._path = path
        self._offset = 0

    def take(self, tag: bytes, size: int | None = None) -> bytes:
        """The next chunk's data. The chunk must carry `tag` and, when `size`
        is given, hold exactly that many bytes."""
        offset = self._offset
        data = self._next(tag)
        if size is not None:
            self._check_size(tag, offset, data, size)
        return data

    def take_counted(
        self, tag: bytes, base_size: int, count_offset: int
    ) -> tuple[bytes, int]:
        """The next chunk's data and child count: a fixed part of `base_size`
        bytes, holding the count at `count_offset`, then one id byte per
        child."""
        offset = self._offset
        data = self._next(tag)
        if len(data) < base_size:
            raise self._error(
                f"{_tag_name(tag)} chunk at {offset:#x} is {len(data)} bytes, "
                f"expected at least {base_size}"
            )
        count = data[count_offset]
        self._check_size(tag, offset, data, base_size + count)
        return data, count

    def finish(self) -> None:
        """Fail if anything follows the last chunk taken."""
        if self._offset != len(self._buffer):
            raise self._error(
                f"Unexpected data at {self._offset:#x} after the S03 chunk"
            )

    def _next(self, tag: bytes) -> bytes:
        offset = self._offset
        buffer = self._buffer
        if offset == len(buffer):
            raise self._error(
                f"Expected {_tag_name(tag)} chunk at {offset:#x}, "
                "found the end of the file"
            )
        if offset + CHUNK_HEADER.size > len(buffer):
            raise self._error(f"Chunk header at {offset:#x} is cut short")
        found, size = CHUNK_HEADER.unpack_from(buffer, offset)
        if found != tag:
            raise self._error(
                f"Expected {_tag_name(tag)} chunk at {offset:#x}, "
                f"found {_tag_name(found)}"
            )
        start = offset + CHUNK_HEADER.size
        end = start + size
        if end > len(buffer):
            raise self._error(
                f"{_tag_name(tag)} chunk at {offset:#x} runs past the end of "
                "the file"
            )
        self._offset = end
        return buffer[start:end]

    def _check_size(self, tag: bytes, offset: int, data: bytes, size: int) -> None:
        if len(data) != size:
            raise self._error(
                f"{_tag_name(tag)} chunk at {offset:#x} is {len(data)} bytes, "
                f"expected {size}"
            )

    def _error(self, reason: str) -> InvalidLgeError:
        return InvalidLgeError(reason, self._path)
